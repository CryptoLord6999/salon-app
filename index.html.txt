    <script>
        const tg = window.Telegram.WebApp;
        tg.expand(); // Раскрыть на весь экран

        let selectedService = null;
        let selectedTime = null;

        function selectService(el, name, price) {
            document.querySelectorAll('.service-item').forEach(i => i.classList.remove('selected'));
            el.classList.add('selected');
            selectedService = { name, price };
            checkForm();
        }

        function selectTime(el, time) {
            document.querySelectorAll('.time-slot').forEach(i => i.classList.remove('selected'));
            el.classList.add('selected');
            selectedTime = time;
            checkForm();
        }

        function checkForm() {
            const btn = document.getElementById('main-btn');
            if (selectedService && selectedTime) {
                btn.style.display = 'block';
                tg.MainButton.show();
                tg.MainButton.setText(`Записаться на ${selectedService.name} в ${selectedTime}`);
            }
        }

        function sendData() {
            const name = document.getElementById('client-name').value || "Аноним";
            
            const data = {
                service: selectedService.name,
                price: selectedService.price,
                time: selectedTime,
                name: name
            };

            // Отправляем данные боту
            tg.sendData(JSON.stringify(data));
        }

        // Обработка нажатия на главную кнопку Telegram (внизу)
        tg.MainButton.onClick(function() {
            sendData();
        });
    </script>
